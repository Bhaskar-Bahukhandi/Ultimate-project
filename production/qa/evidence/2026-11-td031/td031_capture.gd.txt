## TD-031 AC-7 capture (live renderer): the slime fight's first frame through the real hand-off path, as T13a's
## capture did. Run once on HEAD's combat_arena.gd (before) and once on the refactored file (after).
## Run: Godot (not headless) --path AethelgardPrototype --resolution 1280x720 --script <this> -- --out=<png path>

const ARENA := "res://scenes/combat/combat_arena.tscn"
const MEETING := "res://scenes/chapter1/elara_meeting.tscn"

var _out := ""


func _initialize() -> void:
	for a in OS.get_cmdline_user_args():
		if a.begins_with("--out="):
			_out = a.substr(6)
	_run.call_deferred()


func _wait(seconds: float) -> void:
	var end := Time.get_ticks_msec() + int(seconds * 1000.0)
	while Time.get_ticks_msec() < end:
		await process_frame


func _run() -> void:
	await process_frame
	if not OS.get_user_data_dir().contains("aeth_phase0_test"):
		push_error("not isolated")
		quit(2)
		return
	var gm := get_root().get_node("GameManager")
	gm.reset_game()
	gm.set_meta("story_fight_id", "ch1_slime")
	gm.set_meta("story_fight_return_scene", MEETING)
	get_root().get_node("SceneTransitions").change_scene(ARENA)
	await _wait(3.0)
	var arena := current_scene
	print("SCENE ", arena.scene_file_path if arena else "none")
	print("SLIME ", arena.get_node_or_null("Enemies/StoryFightSlime") != null,
		" BACKDROP ", arena.get_node_or_null("StoryFightBackdrop") != null,
		" TREES ", arena.get_node("StoryFightBackdrop").get_child_count() / 2 if arena.has_node("StoryFightBackdrop") else -1)
	await RenderingServer.frame_post_draw
	var img := get_root().get_texture().get_image()
	print("SHOT %s err=%d size=%s" % [_out, img.save_png(_out), img.get_size()])
	quit(0)
