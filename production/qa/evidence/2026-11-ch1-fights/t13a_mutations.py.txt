def run(cmd, **kw):
    return subprocess.run(cmd, capture_output=True, text=True, encoding="utf-8", errors="replace", **kw)


def rd(rel):
    return open(PROJ + "/" + rel, "rb").read()


def edit(rel, old, new):
    raw = rd(rel).decode("utf-8")
    nl = "\r\n" if "\r\n" in raw else "\n"
    s = raw.replace("\r\n", "\n")
    assert s.count(old) == 1, (rel, old[:70], s.count(old))
    open(PROJ + "/" + rel, "wb").write(s.replace(old, new).replace("\n", nl).encode("utf-8"))


def suite(spec):
    path, tag = spec
    env = dict(os.environ)
    env.pop("GODOT", None)
    t0 = time.time()
    r = run(["powershell", "-NoProfile", "-File", PROJ + "/tests/run_phase0_tests.ps1", "-Script", path], env=env)
    out = r.stdout + r.stderr
    res = [l.strip() for l in out.splitlines() if l.startswith(tag)]
    fails, seen = [], set()
    for l in out.splitlines():
        t = l.strip()
        if t.startswith("FAIL: ") and t[6:] not in seen:
            seen.add(t[6:])
            fails.append(t[6:])
        if (t.startswith("RUNNER FAIL") or "SCRIPT ERROR" in t or "Parse Error" in t) and t not in seen:
            seen.add(t)
            fails.append(t)
    return r.returncode, (res[-1] if res else "no result line"), fails, int(time.time() - t0)
