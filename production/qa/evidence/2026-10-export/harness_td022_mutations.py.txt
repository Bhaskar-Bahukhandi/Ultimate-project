def hold_lock(target_expr, seconds, after="", anchor=None):
    """Inserted after `anchor` (default: the forced timeout): another powershell process opens target_expr exclusively
    for `seconds` (then runs `after`), and this script confirms the lock is in place before going on (LOCK: held=True)."""
    anchor = anchor or FORCE_TIMEOUT[1]
    return (anchor, anchor
            + f"        $lockTarget = {target_expr}\n"
            + "        $hold = \"`$f = [IO.File]::Open('$lockTarget', 'OpenOrCreate', 'ReadWrite', 'None'); "
            + f"Start-Sleep -Seconds {seconds}; `$f.Close(); {after}\"\n"
            + "        $null = Start-Process powershell -WindowStyle Hidden -ArgumentList ('-NoProfile -EncodedCommand ' + "
            + "[Convert]::ToBase64String([Text.Encoding]::Unicode.GetBytes($hold)))\n"
            + "        $locked = $false\n"
            + "        foreach ($i in 1..40) { try { ([IO.File]::Open($lockTarget, 'OpenOrCreate', 'ReadWrite', 'None')).Close(); "
            + "Start-Sleep -Milliseconds 100 } catch { $locked = $true; break } }\n"
            + "        Write-Host \"LOCK: held=$locked on $lockTarget\"\n")

def ps(args, timeout=1500):
    return subprocess.run(["powershell", "-NoProfile", "-Command", args], capture_output=True, text=True,
                          encoding="utf-8", errors="replace", timeout=timeout)


def procs():
    return ps("(Get-Process Aethelgard* -ErrorAction SilentlyContinue | Measure-Object).Count").stdout.strip()


def orphans():
    return sum(1 for n in os.listdir(TMP) if re.fullmatch(r"aeth_phase0_test_[0-9a-f]{32}", n))


def main():
    base = open(SRC, "rb").read().decode("utf-8")
    nl = "\r\n" if "\r\n" in base else "\n"
    base = base.replace("\r\n", "\n")
    results = []
    only = set(sys.argv[1:])   # optional subset, e.g. "p q r"; their summary lines replace the old ones
    try:
        for name, flags, edits, need, extra in MUTS:
            if only and name not in only:
                continue
            s = base
            for old, new in edits:
                assert s.count(old) == 1, (name, old[:60], s.count(old))
                s = s.replace(old, new)
            open(MUT, "wb").write(s.replace("\n", nl).encode("utf-8"))
            pre, pre_dirs = procs(), orphans()
            t0 = time.time()
            cmd = f"Set-Location '{PROJ}'; & powershell -NoProfile -File tools/export_windows_mut.ps1 {' '.join(flags)} 2>&1; 'EXIT ' + $LASTEXITCODE"
            r = ps(cmd)
            out = r.stdout + r.stderr
            time.sleep(2)
            post, post_dirs = procs(), orphans()
            if extra.get("cleanup"):
                ps("Get-Process Aethelgard* -ErrorAction SilentlyContinue | Stop-Process -Force; Start-Sleep 2")
            post_clean = procs()
            failed_part = out.split("FAILED:")[-1] if "FAILED:" in out else ""
            if extra.get("expect_pass"):   # the reverted code must let this run through
                ok = "FAILED:" not in out and "EXIT 0" in out
            else:
                ok = need in failed_part and "EXIT 1" in out
            if extra.get("no_ps_error"):
                ok &= not re.search(r"At line|Exception|CategoryInfo|FullyQualifiedErrorId", out)
            if extra.get("see_report"):
                ok &= "report.md" in failed_part
            if extra.get("diag"):
                ok &= "DIAG t+10" in out
            if extra.get("absent"):
                ok &= extra["absent"] not in failed_part
            if extra.get("fallback_killed"):
                ok &= "HasExited=True" in out
            if extra.get("no_orphan"):
                ok &= post_dirs == pre_dirs
            if extra.get("taskkill_exit"):
                ok &= f"STOPTREE: taskkill exit {extra['taskkill_exit']};" in out
            for want in extra.get("out_has", []):
                ok &= want in out
            for unwanted in extra.get("out_lacks", []):
                ok &= unwanted not in out
            smoke_note = ""
            if "file_has" in extra or "file_lacks" in extra:
                time.sleep(extra.get("settle", 0))
                smoke = open(PROJ + "exports/windows/smoke.log", encoding="utf-8", errors="replace").read()
                marker = extra.get("file_has") or extra.get("file_lacks")
                present = marker in smoke
                ok &= present if "file_has" in extra else not present
                smoke_note = f"\nexports/windows/smoke.log read by the harness afterwards: {marker!r} {'present' if present else 'absent'}"
            edits_txt = (f"  script under test sha256 (working tree) {hashlib.sha256(open(SRC, 'rb').read()).hexdigest()}\n"
                         f"  mutant sha256           {hashlib.sha256(open(MUT, 'rb').read()).hexdigest()}\n") + "\n".join(f"  edit {i + 1}:\n    OLD: {o.rstrip()}\n    NEW: {n.rstrip() or '(deleted)'}" for i, (o, n) in enumerate(edits))
            head = (f"mutation ({name}) flags={' '.join(flags) or 'release'}; Aethelgard* processes before {pre}, after {post}, "
                    f"after cleanup {post_clean}; aeth_phase0_test_<guid> folders in %TEMP% before {pre_dirs}, after {post_dirs}; "
                    f"{int(time.time() - t0)} s\n"
                    + ("expected NO failure and exit 0 (the reverted code lets the run through)" if extra.get("expect_pass")
                       else f"expected the FAILED list to contain: {need!r}")
                    + (f"; and not {extra['absent']!r}" if extra.get("absent") else "")
                    + (" ; and FALLBACK HasExited=True" if extra.get("fallback_killed") else "")
                    + (" ; and no new orphan folder" if extra.get("no_orphan") else "")
                    + (" ; and no PowerShell error text" if extra.get("no_ps_error") else "")
                    + (f" ; and STOPTREE taskkill exit {extra['taskkill_exit']}" if extra.get("taskkill_exit") else "")
                    + "".join(f" ; and the output contains {w!r}" for w in extra.get("out_has", []))
                    + "".join(f" ; and the output lacks {w!r}" for w in extra.get("out_lacks", []))
                    + (f" ; and smoke.log contains {extra['file_has']!r}" if "file_has" in extra else "")
                    + (f" ; and smoke.log lacks {extra['file_lacks']!r}" if "file_lacks" in extra else "")
                    + f" -> {'CAUGHT' if ok else 'NOT AS EXPECTED'}{smoke_note}\n{edits_txt}\n--- script output (stdout and stderr) ---\n")
            open(EV + f"ac5_{name}.log", "w", encoding="utf-8", newline="\n").write(head + out)
            results.append((name, ok, pre, post, pre_dirs, post_dirs))
            print(head.splitlines()[0], "->", "CAUGHT" if ok else "NOT AS EXPECTED", flush=True)
            ps("Get-Process Aethelgard* -ErrorAction SilentlyContinue | Stop-Process -Force")
    finally:
        if os.path.exists(MUT):
            os.remove(MUT)
    lines = {}
    if only and os.path.exists(EV + "ac5_summary.log"):
        for l in open(EV + "ac5_summary.log", encoding="utf-8"):
            m = re.match(r"\((\w)\) ", l)
            if m:
                lines[m.group(1)] = l.rstrip("\n")
    for n, ok, pre, post, pd, qd in results:
        lines[n] = (f"({n}) {'CAUGHT' if ok else 'NOT AS EXPECTED'}; Aethelgard* processes before {pre}, after {post}; "
                    f"orphan user:// folders before {pd}, after {qd}" + (" (re-run on its own)" if only else ""))
    all_ok = all("CAUGHT;" in l and "NOT AS EXPECTED" not in l for l in lines.values())
    open(EV + "ac5_summary.log", "w", encoding="utf-8", newline="\n").write(
        "\n".join(lines[k] for k in sorted(lines)) + "\nALL CAUGHT: " + ("yes" if all_ok else "NO") + "\n")
    print("ALL CAUGHT:", all_ok)


main()
